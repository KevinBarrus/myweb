Problem link: [437. Path Sum III - LeetCode](https://leetcode.cn/problems/path-sum-iii/)

Given the root node `root` of a binary tree and an integer `targetSum`, find the number of **paths** in the binary tree whose node values sum to `targetSum`.

A **path** does not need to start at the root or end at a leaf, but its direction must be downward, from parent nodes to child nodes only.

**Example 1:**

![img](https://assets.leetcode.com/uploads/2021/04/09/pathsum3-1-tree.jpg)

```
Input: root = [10,5,-3,3,2,null,11,3,-2,null,1], targetSum = 8
Output: 3
Explanation: There are 3 paths whose sum is 8, as shown in the figure.
```

**Example 2:**

```
Input: root = [5,4,8,11,null,13,4,7,2,null,null,5,1], targetSum = 22
Output: 3
```

**Constraints:**

- The number of nodes in the binary tree is in the range `[0,1000]`.
- `-109 <= Node.val <= 109`
- `-1000 <= targetSum <= 1000`

Each path can be regarded as an array, so the problem becomes counting the number of subsequences in an array whose sum is `target`. Clearly, use prefix sums.

First, to obtain the node values traversed by every path, use depth-first search.

When we reach node `node`, calculate its prefix sum `curr`. We want a node `node1` to exist between the root node `root` and `node` such that the sum from `node1` to `node` is `target`. This means that the prefix sum of `node1` is `curr-target`.

Use a hash table to record the counts. The key is a prefix sum and the value is its occurrence count.

When reaching node `node`, check whether the key `curr-target` exists in the hash table. If it does, record the number of times this prefix sum has appeared on the current path. Then recursively count the occurrences of the target value in the left and right subtrees. Finally, backtrack and enter the other branch.

C++

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    unordered_map<uint64_t,int>cnt;//key: prefix sum; value: occurrence count
    int dfs(TreeNode*root,uint64_t curr,int target) {
        if(!root) return 0;
        int res=0;
        curr+=root->val;
        if(cnt.count(curr-target)) {
            //if the key curr-target exists
            res=cnt[curr-target];//res is the number of occurrences on this path
        }
        cnt[curr]++;
        res+=dfs(root->left,curr,target);
        res+=dfs(root->right,curr,target);
        cnt[curr]--;//backtrack
        return res;
    }
    int pathSum(TreeNode* root, int targetSum) {
        cnt[0]=1;//for the root node, the prefix sum is 0, so count one path
        return dfs(root,0,targetSum);
    }
};
```

Java
```java
import java.util.HashMap;
import java.util.Map;

class Solution {

    public static class TreeNode {
        int val;
        TreeNode left;
        TreeNode right;
        TreeNode() {};
        TreeNode(int val) {
            this.val = val;
        }
        TreeNode(int val, TreeNode left, TreeNode right) {
            this.val = val;
            this.left = left;
            this.right = right;
        }
    }

    private int ans;

    public int pathSum(TreeNode root, int targetSum) {
        //key: path sum from the root node to this node
        //value: occurrence count of this path sum
        Map<Long,Integer> cnt = new HashMap();
        cnt.put(0L,1);
        dfs(root,0L,targetSum,cnt);
        return ans;
    }

    //s is the path sum from the root node to the current node (excluding the current node)
    private void dfs(TreeNode node, long s, int targetSum, Map<Long,Integer> cnt) {
        if(node == null) {
            return;
        }

        s += node.val;
        ans += cnt.getOrDefault(s - targetSum, 0); //prefix sum
        cnt.merge(s, 1, Integer::sum); //cnt[s]++

        //what if merge is called before ans is increased? It only causes a problem when targetSum is 0, because then ans += cnt[s]
        //if targetSum is 0 and the root value is 1, first increasing cnt[1] by 1 would make ans 1, but no path has sum 0
        dfs(node.left, s, targetSum, cnt);
        dfs(node.right, s, targetSum, cnt);
        //backtrack
        cnt.merge(s, -1, Integer::sum); //all child nodes of this node have been counted; its job is complete
    }
}
```

Go

```go
package main

type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

func pathSum(root *TreeNode, targetSum int) (ans int) {
    cnt := map[int]int{0:1}
    var dfs func(*TreeNode, int)
    dfs = func(node *TreeNode, s int) {
        if node == nil {
            return
        }
        s += node.Val
        ans += cnt[s - targetSum]
        cnt[s]++
        dfs(node.Left, s)
        dfs(node.Right, s)
        cnt[s]--
    }
    dfs(root, 0)
    return ans
}
```
