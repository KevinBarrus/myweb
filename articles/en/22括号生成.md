[22. Generate Parentheses - LeetCode](https://leetcode.cn/problems/generate-parentheses/description/?envType=study-plan-v2&envId=top-100-liked)

The integer `n` represents the number of pairs of parentheses. Design a function that generates all possible **valid** combinations of parentheses.

**Example 1:**

**Input:** n = 3
**Output:** ["((()))","(()())","(())()","()(())","()()()"]

**Example 2:**

**Input:** n = 1
**Output:** ["()"]

**Constraints:**

- `1 <= n <= 8`

Essentially, choose `n` positions from `0, 1, 2, ... , 2n - 1` to fill with left parentheses, and fill the remaining `n` positions with right parentheses. Notice that for any prefix of this string, the number of right parentheses must not exceed the number of left parentheses. If there are more left parentheses, right parentheses can still be added later; if there are more right parentheses in a prefix, adding left parentheses later cannot form a pair.

Therefore, for each position, the problem becomes whether to choose: fill it with a left parenthesis or a right parenthesis. If the current counts of left and right parentheses are equal, we must fill a left parenthesis. If the number of right parentheses is smaller than the number of left parentheses, we can fill a right parenthesis. Since both counts are initially 0, this strategy necessarily fills the first position with a left parenthesis, and it is clear that the number of right parentheses cannot exceed the number of left parentheses later. This is valid.

```python
class Solution:
    def generateParenthesis(self, n: int) -> List[str]:
        ans = []
        path = [''] * (n * 2) # n left parentheses, n right parentheses

        # left: number of left parentheses, right: number of right parentheses
        def dfs(left: int, right:int) -> None:
            if right == n: # all 2n parentheses have been filled
                ans.append(''.join(path))
                return
            if left < n:
                # the number of left parentheses has not reached n, so we can fill one
                path[left + right] = '('
                dfs(left + 1, right)
            if right < left:
                path[left + right] = ')'
                dfs(left, right + 1)

        dfs(0, 0)
        return ans
```

There is no need to restore the state here, because the element at position `left + right` is overwritten directly.

```python
from typing import List

def generateParenthesis(n: int) -> List[str]:
    ans = []
    path = [''] * (n * 2)

    # left: number of left parentheses, right: number of right parentheses
    def dfs(left: int, right: int) -> None:
        if right == n:
            # filling is complete
            ans.append("".join(path))
            return
        if left < n:
            # a left parenthesis can be filled
            path[left + right] = '(' # overwritten directly, so path does not need to be cleared after filling
            dfs(left + 1, right)
        if right < left:
            # a right parenthesis can be filled
            path[left + right] = ')'
            dfs(left, right + 1)
    dfs(0, 0)
    return ans

def main():
    with open("input.txt", "r") as f:
        nums = f.read().split()
    # iterate over n in input.txt
    for num in nums:
        n = int(num)
        result = generateParenthesis(n)
        # print the result
        print(f"n = {n}")
        print(result)
        print()

if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedReader;
import java.io.FileReader;
import java.util.ArrayList;
import java.util.List;

public class main {
    static int n;
    static List<String> ans;
    static char[] path;

    public static List<String> generateParenthesis(int n) {
        main.n = n;
        ans = new ArrayList<>();
        path = new char[n * 2];
        dfs(0, 0);
        return ans;
    }

    public static void dfs(int left, int right) {
        if(right == n) {
            // filling is complete
            ans.add(new String(path));
            return;
        }

        if(left < n) {
            path[left + right] = '(';
            dfs(left + 1, right);
        }

        if(right < left) {
            path[left + right] = ')';
            dfs(left, right + 1);
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new FileReader("input.txt"));
        StringBuilder sb = new StringBuilder();
        String line;

        while((line = br.readLine()) != null) {
            if(line.isEmpty()) {
                continue;
            }

            int n = Integer.parseInt(line.trim());
            List<String> result = generateParenthesis(n);

            sb.append("n = ").append(n).append("\n");
            sb.append(result).append("\n\n");
        }
        System.out.println(sb);
    }
}
```

```go
package main

import (
    "fmt"
    "os"
    "strconv"
    "strings"
)
var ans []string
var path []byte

func dfs(n int, left int, right int) {
    if right == n {
        ans = append(ans, string(path))
        return
    }

    if left < n {
        path[left+right] = '('
        dfs(n, left+1, right)
    }

    if right < left {
        path[left+right] = ')'
        dfs(n, left, right+1)
    }
}

func generateParenthesis(n int) []string {
    ans = nil
    path = make([]byte, n*2)
    dfs(n, 0, 0)
    return ans
}

func main() {
    data, _ := os.ReadFile("input.txt")
    nums := strings.Fields(string(data))
    for _, s := range nums {
        n, _ := strconv.Atoi(s)
        result := generateParenthesis(n)
        fmt.Printf("n = %d\n", n)
        fmt.Println(result)
        fmt.Println()
    }
}
```

```c++
#include<iostream>
#include<vector>
#include<string>
#include<fstream>
using namespace std;

vector<string>ans;
string path;

void dfs(int n, int left, int right) {
    if(right == n) {
        ans.emplace_back(path);
        return;
    }

    if(left < n) {
        path[left + right] = '(';
        dfs(n, left + 1, right);
    }

    if(right < left) {
        path[left + right] = ')';
        dfs(n, left, right + 1);
    }
}

vector<string> generateParenthesis(int n) {
    ans.clear();
    path = string(n * 2,' ');
    dfs(n, 0, 0);
    return ans;
}

int main() {
    ifstream ifs("input.txt");
    int n;

    while(ifs >> n) {
        auto result = generateParenthesis(n);
        cout << "n = " << n << endl;
        cout << "[";
        for(int i = 0; i < result.size(); i++) {
            cout << result[i];
            if(i != result.size() - 1) {
                cout << ",";
            }
        }
        cout << "]" << endl << endl;
    }
    return 0;
}
```

```typescript
import * as fs from "fs";

function generateParenthesis(n: number) :string[] {
    let ans:string[] = [];
    let path:string[] = new Array(n * 2);

    function dfs(left: number, right: number) {
        if(right == n) {
            ans.push(path.join(""));
            return;
        }
        if(left < n) {
            path[left + right] = '(';
            dfs(left + 1, right);
```
